if判斷式: 滿足某個條件，才執行程式
基礎語法
if(布林值){

}
程式範例
if(true){
    System.out.println("Hello");
}

int x=1000;
if(x>2000){
    System.out.println("Hello");
}

if...else正反結構
基礎語法
if(布林值){
    //如果布林值是true，執行此程式碼區塊
}else{
    //如果布林值是false，執行此程式碼區塊
}

程式範例
int x=1000;
if(x>2000){
    System.out.println(">2000");
}else{
    System.out.println("<=2000");
}

if...else if...else多條件結構
基礎語法
if(布林值){
    //對應布林值是true，執行此程式碼區塊
}else if(布林值){
    //對應布林值是true，執行此程式碼區塊
}else{
    //上方布林值都是false，執行此程式碼區塊
}

程式範例
int x=1000;
if(x>2000){
    System.out.println(">2000");
}else if(x=2000){
    System.out.println("=2000");
}else{
    System.out.println("<2000");
}

String name=Serena;
if(name==Mike){
    System.out.println("I am Mike.");
}else if(name==Serena){
    System.out.println("I am Serena.");
}elseif(name==Linda){
    System.out.println("I am Linda.");
}


switch判斷式: 適合處理多條件情況
基礎語法
swith(變數名稱){
    case 比對的資料:
        //變數中的資料和比對到任何資料相同
        //執行這裡的程式
        break;
    case 比對的資料:
        //變數中的資料和比對到任何資料相同
        //執行這裡的程式
        break;
    default:
        //變數中的資料沒有比對到任何資料
        //執行這裡的程式
        break;    //強迫程式跳出
}
//可以放很多很多個case

程式範例
String name="Serena"
swith(name){
    case "Mike":
        System.out.println("I am Mike.");
        break;  
    case "Serena":
        /System.out.println("I am Serena.");
        break;
    default:
        System.out.println("Not Mike or Serena");
        break;
}
